        function toggleGlass(index) {

            const glasses = document.querySelectorAll('.glass');

            if (index < waterCount) {

                for (let i = waterCount - 1; i >= index; i--) {

                    glasses[i].classList.remove('filled');

                }

                waterCount = index;

            } else {

                for (let i = waterCount; i <= index; i++) {

                    glasses[i].classList.add('filled');

                }

                waterCount = index + 1;

            }

            updateHydrationDisplay();

            lastWaterTime = new Date();

        }

 

        function addWater() {

            if (waterCount < 8) {

                const glasses = document.querySelectorAll('.glass');

                glasses[waterCount].classList.add('filled');

                lastWaterTime = new Date();

               

                if (waterCount === 8) {

                    hydrationStreak++;

                }

            } else {

                showNotification('You already completed your daily goal! 🎉');

            }

        }
